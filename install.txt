# Nextiva Extensions installer (PowerShell) - installs or updates one extension, or all of them.
# Run from Win+R (or a .cmd) with:
#   powershell -NoProfile -ExecutionPolicy Bypass -Command "$x='row-id-copier'; irm https://jasonkbarry.github.io/extensions/install.txt | iex"
# $x = an extension id from extensions.json, or 'all'.
$ErrorActionPreference = 'Stop'
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
$Site = 'https://jasonkbarry.github.io/extensions'
$Base = Join-Path $env:LOCALAPPDATA 'Nextiva Extensions'
function Say($t, $c = 'Gray') { Write-Host $t -ForegroundColor $c }
function NoCache($u) { $u + '?t=' + [DateTime]::Now.Ticks }

# Every unpacked extension folder loaded in any Chrome / Edge profile on this computer
function Get-LoadedFolders {
  $list = New-Object System.Collections.Generic.List[string]
  foreach ($root in @("$env:LOCALAPPDATA\Google\Chrome\User Data", "$env:LOCALAPPDATA\Microsoft\Edge\User Data")) {
    if (-not (Test-Path $root)) { continue }
    foreach ($prof in (Get-ChildItem $root -Directory -ErrorAction SilentlyContinue)) {
      foreach ($pf in @('Preferences', 'Secure Preferences')) {
        $file = Join-Path $prof.FullName $pf
        if (-not (Test-Path $file)) { continue }
        try { $text = [IO.File]::ReadAllText($file) } catch { continue }
        foreach ($m in [regex]::Matches($text, '"path"\s*:\s*"([A-Za-z]:\\\\[^"]+)"')) {
          $p = $m.Groups[1].Value -replace '\\\\', '\'
          if ((Test-Path (Join-Path $p 'manifest.json')) -and -not $list.Contains($p)) { $list.Add($p) }
        }
      }
    }
  }
  return $list
}

function Install-One($id, $loaded) {
  $info = Invoke-RestMethod -UseBasicParsing -Uri (NoCache "$Site/$id.json")
  Say ''
  Say ("  " + $info.title + "  (latest " + $info.version + ")") 'Cyan'

  # Where is it installed? Match the manifest name (names may end with a version, so match the start).
  $targets = New-Object System.Collections.Generic.List[string]
  foreach ($p in $loaded) {
    try { $n = (Get-Content (Join-Path $p 'manifest.json') -Raw | ConvertFrom-Json).name } catch { continue }
    if ($n -like ($info.match + '*') -and -not $targets.Contains($p)) { $targets.Add($p) }
  }
  $default = Join-Path $Base $info.folder
  if ((Test-Path (Join-Path $default 'manifest.json')) -and -not $targets.Contains($default)) { $targets.Add($default) }
  $isNew = $targets.Count -eq 0
  if ($isNew) { $targets.Add($default) }

  # Download and unpack
  $tmp = Join-Path $env:TEMP ('nxe-' + [Guid]::NewGuid())
  New-Item -ItemType Directory -Path $tmp | Out-Null
  $zip = Join-Path $tmp 'ext.zip'
  Invoke-WebRequest -UseBasicParsing -Uri ("$Site/" + $info.zip) -OutFile $zip
  Expand-Archive -Path $zip -DestinationPath $tmp -Force
  $src = Join-Path $tmp $info.folder
  if (-not (Test-Path (Join-Path $src 'manifest.json'))) { throw ("The download for " + $info.title + " did not contain the extension.") }

  # Copy into each install folder; manifest.json last so a self-reloading extension only reloads once the new code is in place
  foreach ($t in $targets) {
    $old = '(new)'
    if (Test-Path (Join-Path $t 'manifest.json')) { try { $old = (Get-Content (Join-Path $t 'manifest.json') -Raw | ConvertFrom-Json).version } catch {} }
    New-Item -ItemType Directory -Path $t -Force | Out-Null
    Get-ChildItem $t -Filter 'VERSION*' -File -ErrorAction SilentlyContinue | Remove-Item -Force
    robocopy "$src" "$t" /E /XF manifest.json /R:2 /W:1 /NFL /NDL /NJH /NJS /NP | Out-Null
    if ($LASTEXITCODE -ge 8) { throw "Could not copy files to $t (robocopy code $LASTEXITCODE). Is the folder open or read-only?" }
    Copy-Item (Join-Path $src 'manifest.json') -Destination $t -Force
    Say ("    $t  ($old -> " + $info.version + ")") 'Green'
  }
  Remove-Item $tmp -Recurse -Force -ErrorAction SilentlyContinue
  return [pscustomobject]@{ Info = $info; IsNew = $isNew; Folder = $targets[0] }
}

Say ''
Say '  Nextiva Extensions installer' 'Cyan'
Say '  ----------------------------' 'Cyan'
try {
  if (-not $x) { throw "No extension chosen. Use the command from the extension's page on $Site" }
  $ids = if ($x -eq 'all') { (Invoke-RestMethod -UseBasicParsing -Uri (NoCache "$Site/extensions.json")).extensions } else { @($x) }
  $loaded = Get-LoadedFolders
  $results = @()
  foreach ($id in $ids) {
    try { $results += Install-One $id $loaded }
    catch { Say ("    Failed: " + $_.Exception.Message) 'Red' }
  }

  $new = @($results | Where-Object { $_.IsNew })
  $upd = @($results | Where-Object { -not $_.IsNew })
  if ($upd.Count) {
    Say ''
    Say '  Updated. To finish:' 'Yellow'
    foreach ($r in $upd) {
      if ($r.Info.selfReload) { Say ("    - " + $r.Info.title + " reloads itself within about a minute.") }
      else { Say ("    - " + $r.Info.title + ": click its reload arrow in chrome://extensions.") }
    }
  }
  if ($new.Count) {
    Say ''
    Say '  First-time setup (one time per extension):' 'Yellow'
    Say '    1. Chrome will open the Extensions page. Turn on "Developer mode" (top right).'
    Say '    2. Click "Load unpacked". In the folder window, click the address bar, press Ctrl+V and Enter, then "Select Folder".'
    Say '    3. Pin the extension from the puzzle-piece menu.'
    if ($new.Count -gt 1) { Say '    Load each of these folders (the first one is on your clipboard):' }
    foreach ($r in $new) { Say ("       " + $r.Folder) }
    Set-Clipboard -Value $new[0].Folder
    try { Start-Process 'chrome.exe' 'chrome://extensions' } catch { Say '    (Open chrome://extensions yourself - Chrome was not found.)' }
  }
} catch {
  Say ''
  Say ('  Install failed: ' + $_.Exception.Message) 'Red'
  Say ("  You can also download the zips from $Site") 'Red'
}
Say ''
[void](Read-Host '  Press Enter to close')
